-- ============================================================
-- FILE 01: Setup, create tables, load CSV from stage
-- ============================================================

-- Create database and schemas
CREATE DATABASE IF NOT EXISTS FRAUD_ANALYTICS;
USE DATABASE FRAUD_ANALYTICS;
CREATE SCHEMA IF NOT EXISTS RAW;
CREATE SCHEMA IF NOT EXISTS ANALYTICS;
CREATE SCHEMA IF NOT EXISTS REPORTING;

-- Raw transactions table (22 columns matching fraudTest.csv)
CREATE OR REPLACE TABLE RAW.TRANSACTIONS (
    TRANS_DATE_TRANS_TIME   TIMESTAMP,
    CC_NUM                  VARCHAR(25),
    MERCHANT                VARCHAR(100),
    CATEGORY                VARCHAR(50),
    AMT                     FLOAT,
    FIRST                   VARCHAR(50),
    LAST                    VARCHAR(50),
    GENDER                  VARCHAR(1),
    STATE                   VARCHAR(2),
    ZIP                     VARCHAR(10),
    LAT                     FLOAT,
    LONG                    FLOAT,
    CITY_POP                INTEGER,
    JOB                     VARCHAR(100),
    DOB                     DATE,
    TRANS_NUM               VARCHAR(50),
    UNIX_TIME               BIGINT,
    MERCH_LAT               FLOAT,
    MERCH_LONG              FLOAT,
    IS_FRAUD                INTEGER
);

-- File format for CSV loading
CREATE OR REPLACE FILE FORMAT FRAUD_ANALYTICS.RAW.CSV_FORMAT
    TYPE = 'CSV'
    FIELD_OPTIONALLY_ENCLOSED_BY = '"'
    SKIP_HEADER = 1
    TIMESTAMP_FORMAT = 'AUTO'
    DATE_FORMAT = 'AUTO'
    NULL_IF = ('NULL', 'null', '');

-- Load from stage
COPY INTO RAW.TRANSACTIONS
FROM @FRAUD_STAGE
FILE_FORMAT = (FORMAT_NAME = 'CSV_FORMAT')
ON_ERROR = 'CONTINUE';

-- Verify: should show ~770K rows, ~0.38% fraud rate
SELECT
    COUNT(*)                                   AS total_records,
    SUM(IS_FRAUD)                              AS total_fraud,
    ROUND(SUM(IS_FRAUD) / COUNT(*) * 100, 4)  AS fraud_rate_pct
FROM RAW.TRANSACTIONS;
-- ============================================================
-- FILE 02: Data quality checks + feature engineering
-- ============================================================

-- Null check across critical columns
SELECT
    SUM(CASE WHEN CC_NUM    IS NULL THEN 1 ELSE 0 END) AS null_cc_num,
    SUM(CASE WHEN AMT       IS NULL THEN 1 ELSE 0 END) AS null_amt,
    SUM(CASE WHEN IS_FRAUD  IS NULL THEN 1 ELSE 0 END) AS null_is_fraud
FROM RAW.TRANSACTIONS;

-- Enriched table with time features, age, and distance
CREATE OR REPLACE TABLE ANALYTICS.TRANSACTIONS_ENRICHED AS
SELECT
    TRANS_NUM, CC_NUM,
    TRANS_DATE_TRANS_TIME                              AS TRANS_TIMESTAMP,
    DATE(TRANS_DATE_TRANS_TIME)                        AS TRANS_DATE,

    -- Time features
    HOUR(TRANS_DATE_TRANS_TIME)                        AS TRANS_HOUR,
    CASE
        WHEN HOUR(TRANS_DATE_TRANS_TIME) BETWEEN 0  AND 5  THEN 'Late Night'
        WHEN HOUR(TRANS_DATE_TRANS_TIME) BETWEEN 6  AND 11 THEN 'Morning'
        WHEN HOUR(TRANS_DATE_TRANS_TIME) BETWEEN 12 AND 17 THEN 'Afternoon'
        ELSE 'Evening'
    END                                                AS TIME_OF_DAY,

    -- Cardholder age
    DATEDIFF('year', DOB, TRANS_DATE_TRANS_TIME)       AS CARDHOLDER_AGE,

    -- Distance: cardholder vs merchant (Haversine formula)
    ROUND(
        3959 * ACOS(LEAST(1.0,
            COS(RADIANS(LAT)) * COS(RADIANS(MERCH_LAT))
            * COS(RADIANS(MERCH_LONG) - RADIANS(LONG))
            + SIN(RADIANS(LAT)) * SIN(RADIANS(MERCH_LAT))
        )), 2
    )                                                  AS DIST_MILES,

    AMT, CATEGORY, MERCHANT, STATE, CITY_POP,
    IS_FRAUD

FROM RAW.TRANSACTIONS
WHERE TRANS_NUM IS NOT NULL AND AMT > 0;
-- ============================================================
-- FILE 03: Z-score + IQR anomaly detection + risk scoring
-- ============================================================

-- Step 1: Per-card baseline (built on LEGITIMATE txns only)
CREATE OR REPLACE TABLE ANALYTICS.CARD_BASELINES AS
SELECT
    CC_NUM,
    COUNT(*)                                           AS TOTAL_TXN_COUNT,
    ROUND(AVG(AMT), 2)                                 AS AVG_TXN_AMT,
    ROUND(STDDEV(AMT), 2)                              AS STDDEV_TXN_AMT,
    ROUND(PERCENTILE_CONT(0.25)
        WITHIN GROUP (ORDER BY AMT), 2)               AS Q1_AMT,
    ROUND(PERCENTILE_CONT(0.75)
        WITHIN GROUP (ORDER BY AMT), 2)               AS Q3_AMT
FROM ANALYTICS.TRANSACTIONS_ENRICHED
WHERE IS_FRAUD = 0   -- legitimate only
GROUP BY CC_NUM
HAVING COUNT(*) >= 5;

-- Step 2: Score every transaction
CREATE OR REPLACE TABLE ANALYTICS.TRANSACTIONS_SCORED AS
WITH scored AS (
    SELECT
        t.*,
        -- Z-score: std deviations above card mean
        CASE WHEN b.STDDEV_TXN_AMT > 0
            THEN ROUND((t.AMT - b.AVG_TXN_AMT)
                      / b.STDDEV_TXN_AMT, 2)
            ELSE 0
        END                                            AS Z_SCORE,

        -- Anomaly flags (each = 1 point toward risk score)
        CASE WHEN (t.AMT - b.AVG_TXN_AMT)
            / NULLIF(b.STDDEV_TXN_AMT,0) > 3
            THEN 1 ELSE 0 END                         AS FLAG_ZSCORE,

        CASE WHEN t.AMT > (b.Q3_AMT
            + 1.5 * (b.Q3_AMT - b.Q1_AMT))
            THEN 1 ELSE 0 END                         AS FLAG_IQR,

        CASE WHEN t.TIME_OF_DAY = 'Late Night'
            THEN 1 ELSE 0 END                         AS FLAG_LATE_NIGHT,

        CASE WHEN t.DIST_MILES > 100
            THEN 1 ELSE 0 END                         AS FLAG_DISTANCE,

        CASE WHEN t.CATEGORY IN (
            'shopping_net','misc_net','grocery_pos')
            THEN 1 ELSE 0 END                         AS FLAG_HIGH_RISK_CAT

    FROM ANALYTICS.TRANSACTIONS_ENRICHED t
    LEFT JOIN ANALYTICS.CARD_BASELINES b ON t.CC_NUM = b.CC_NUM
)
SELECT *,
    (FLAG_ZSCORE + FLAG_IQR + FLAG_LATE_NIGHT
     + FLAG_DISTANCE + FLAG_HIGH_RISK_CAT)             AS RISK_SCORE,
    CASE
        WHEN (FLAG_ZSCORE+FLAG_IQR+FLAG_LATE_NIGHT
              +FLAG_DISTANCE+FLAG_HIGH_RISK_CAT) >= 4 THEN 'CRITICAL'
        WHEN (FLAG_ZSCORE+FLAG_IQR+FLAG_LATE_NIGHT
              +FLAG_DISTANCE+FLAG_HIGH_RISK_CAT) = 3  THEN 'HIGH'
        WHEN (FLAG_ZSCORE+FLAG_IQR+FLAG_LATE_NIGHT
              +FLAG_DISTANCE+FLAG_HIGH_RISK_CAT) = 2  THEN 'MEDIUM'
        WHEN (FLAG_ZSCORE+FLAG_IQR+FLAG_LATE_NIGHT
              +FLAG_DISTANCE+FLAG_HIGH_RISK_CAT) = 1  THEN 'LOW'
        ELSE 'NORMAL'
    END                                                AS RISK_TIER
FROM scored;
-- ============================================================
-- FILE 04: Reporting views — Power BI connects here
-- ============================================================

-- KPI summary card
CREATE OR REPLACE VIEW REPORTING.VW_KPI_SUMMARY AS
SELECT
    COUNT(*)                                           AS TOTAL_TRANSACTIONS,
    SUM(IS_FRAUD)                                      AS TOTAL_FRAUD_CASES,
    ROUND(SUM(IS_FRAUD)::FLOAT / COUNT(*) * 100, 3)  AS OVERALL_FRAUD_RATE_PCT,
    ROUND(SUM(CASE WHEN IS_FRAUD=1 THEN AMT ELSE 0 END),2) AS TOTAL_FRAUD_AMOUNT,
    COUNT(CASE WHEN RISK_TIER = 'CRITICAL' THEN 1 END) AS CRITICAL_ALERTS
FROM ANALYTICS.TRANSACTIONS_SCORED;

-- Fraud by hour (includes RISK_TIER for slicer)
CREATE OR REPLACE VIEW REPORTING.VW_FRAUD_BY_HOUR AS
SELECT
    RISK_TIER, TRANS_HOUR, TIME_OF_DAY,
    COUNT(*)                                           AS TOTAL_TXN,
    SUM(IS_FRAUD)                                      AS FRAUD_COUNT,
    ROUND(SUM(IS_FRAUD)::FLOAT/COUNT(*)*100,2)        AS FRAUD_RATE_PCT
FROM ANALYTICS.TRANSACTIONS_SCORED
GROUP BY RISK_TIER, TRANS_HOUR, TIME_OF_DAY;

-- High risk accounts drill-through
CREATE OR REPLACE VIEW REPORTING.VW_HIGH_RISK_ACCOUNTS AS
SELECT
    CC_NUM,
    COUNT(*)                                           AS TOTAL_TXN,
    SUM(IS_FRAUD)                                      AS CONFIRMED_FRAUD_COUNT,
    COUNT(CASE WHEN RISK_TIER IN
        ('CRITICAL','HIGH') THEN 1 END)               AS HIGH_RISK_FLAGS,
    ROUND(MAX(AMT), 2)                                AS MAX_TXN_AMT,
    ROUND(SUM(AMT), 2)                                AS TOTAL_SPEND,
    MAX(TRANS_DATE)                                    AS LAST_TXN_DATE
FROM ANALYTICS.TRANSACTIONS_SCORED
WHERE RISK_TIER IN ('CRITICAL','HIGH')
GROUP BY CC_NUM
ORDER BY HIGH_RISK_FLAGS DESC
LIMIT 500;
Key Findings From the Data
0.38%
Overall fraud rate across 770K transactions — highly imbalanced dataset requiring statistical rather than count-based detection
shopping_net
Highest fraud rate category at 1.05% — online shopping transactions 2.8× more likely to be fraudulent than average
22–23h
Fraud spikes sharply at hours 22–23 (10–11pm), with elevated rates through early morning (0–4am) — matching known fraud behaviour patterns
$1,000+
Highest absolute fraud volume in the $1,000+ band — large transactions significantly more likely to be flagged across all risk tiers
5,288
Total high-risk flags raised across the dataset — composite scoring isolates the top accounts for priority manual review
